## Задание 1. `.proto` для стриминговой платформы (фильмы, сериалы, подписки)

**Сначала фраза-план (30 секунд, показывает мышление):**
«Домен: контент двух типов — фильмы и сериалы (у сериалов сезоны/эпизоды, у фильмов — нет), тарифы подписок. Значит мне нужны: enum для типа контента и тарифа, `oneof` для «своих» полей фильма/сериала (аналог sealed — нельзя создать сериал с полями фильма), `repeated` для списков, отдельный сервис каталога с постраничной выдачей и server-streaming для ленты. Все деньги — целыми, не float».

```protobuf
syntax = "proto3";
package streaming.v1;

option java_package = "ru.streaming.v1";
option java_multiple_files = true;

// --- Перечисления: ПЕРВОЕ значение всегда 0 = дефолт proto3 ---
enum ContentType {
    CONTENT_TYPE_UNSPECIFIED = 0;
    MOVIE = 1;
    SERIES = 2;
}

enum SubscriptionTier {
    TIER_UNSPECIFIED = 0;
    FREE = 1;      // часть каталога, с рекламой
    BASIC = 2;     // фильмы + 1 экран
    PREMIUM = 3;   // всё + 4K + 4 экрана
}

// --- Деньги: целые минорные единицы, НЕ float ---
message Money {
    int64 minor_units = 1;   // копейки/центы
    string currency = 2;     // "RUB"
}

// --- Домен ---
message Episode {
    int64 id = 1;
    int32 number = 2;
    string title = 3;
    int32 duration_minutes = 4;
}

message Season {
    int32 number = 1;
    repeated Episode episodes = 2;
}

message Content {
    int64 id = 1;
    string title = 2;
    ContentType type = 3;
    int32 release_year = 4;
    repeated string genres = 5;
    SubscriptionTier min_tier = 6;   // минимальный тариф для просмотра

    // «Свои» данные типа: у фильма режиссёр, у сериала сезоны.
    // Аналог sealed: невалидное состояние невыразимо — нет полей-призраков.
    oneof details {
        MovieDetails movie = 7;
        SeriesDetails series = 8;
    }

    reserved 11, 12;        // удалённые поля заняты навсегда
    reserved "rating";
}

message MovieDetails {
    string director = 1;
}
message SeriesDetails {
    repeated Season seasons = 1;
}

message Subscription {
    int64 id = 1;
    SubscriptionTier tier = 2;
    Money monthly_price = 3;
    int32 screen_limit = 4;
}

// --- Запросы/ответы ---
message GetContentRequest  { int64 id = 1; }

message ListContentRequest {
    int32 page_size = 1;
    string page_token = 2;          // курсорная пагинация, не номер страницы
    ContentType type_filter = 3;    // необязательный фильтр (0 = без фильтра)
}
message ListContentResponse {
    repeated Content items = 1;
    string next_page_token = 2;
}

message SearchRequest  { string query = 1; SubscriptionTier allowed_tier = 2; }
message SubscribeRequest { int64 user_id = 1; SubscriptionTier tier = 2; }

// --- Сервис ---
service StreamingService {
    rpc GetContent (GetContentRequest) returns (Content);        // unary
    rpc ListContent (ListContentRequest) returns (ListContentResponse); // unary, постранично
    rpc Search (SearchRequest) returns (stream Content);         // server streaming: лента результатов
    rpc Subscribe (SubscribeRequest) returns (Subscription);     // unary
}
```

**Что проговорить при защите (галочки экзаменатора):**
- `syntax = "proto3"` первой строкой; `package` = пространство имён.
- Enum с 0-UNSPECIFIED — требование proto3: 0 = дефолт, совместимость со старыми данными.
- `oneof` вместо nullable-призраков — связь с sealed/enum (тот же принцип, на уровне полей).
- Новое поле = новый номер → совместимо; удалённое → `reserved` (номер + имя).
- Деньги целыми: float для денег нельзя (ошибки округления binary floating point).
- Пагинация токеном (курсором): вставка данных между страницами не ломает выдачу.
- `stream Content` — server streaming: объяснить 4 типа вызовов, какой где был бы нужен (bidi — чат поддержки).

## Задание 2. GraphQL-схема для ресторана (меню, заказы, курьеры)

**Фраза-план:** «Домен: меню, клиенты, курьеры, заказы с жизненным циклом. Заказ без курьера — валидное состояние, поэтому courier nullable осознанно. Живой трекинг статуса — Subscription (push вместо polling). Мутации принимают input, фильтры — input-объект, деньги — целым».

```graphql
# --- Меню ---
enum Category { STARTER MAIN DESSERT DRINK }

type Money {
    minorUnits: Int!      # копейки — целым, не Float!
    currency: String!
}

type Dish {
    id: ID!
    name: String!
    description: String           # nullable: описания может не быть
    price: Money!
    category: Category!
    isAvailable: Boolean!
    spicyLevel: Int               # nullable: не для всех блюд имеет смысл
}

# --- Люди ---
type Customer {
    id: ID!
    name: String!
    phone: String!
    addresses: [String!]!
}

type Courier {
    id: ID!
    name: String!
    phone: String!
    isOnShift: Boolean!
    activeOrders: [Order!]!       # связь: DataLoader, иначе N+1
}

# --- Заказы ---
enum OrderStatus { NEW COOKING DELIVERING DELIVERED CANCELLED }

type OrderItem {
    dish: Dish!
    quantity: Int!
}

type Order {
    id: ID!
    customer: Customer!
    items: [OrderItem!]!
    total: Money!
    status: OrderStatus!
    courier: Courier              # NULLABLE ОСОЗНАННО: «ещё не назначен» — валидное состояние
    createdAt: String!
}

# --- Входы мутаций (только input!) ---
input OrderItemInput {
    dishId: ID!
    quantity: Int!
}

input CreateOrderInput {
    customerId: ID!
    address: String!
    items: [OrderItemInput!]!
}

input DishFilter {
    category: Category
    maxPriceMinor: Int
    onlyAvailable: Boolean = false
}

type Query {
    menu(filter: DishFilter): [Dish!]!
    dish(id: ID!): Dish                 # nullable: 404 выражается null
    order(id: ID!): Order
    couriers(onShift: Boolean): [Courier!]!
    ordersByStatus(status: OrderStatus!): [Order!]!
}

type Mutation {
    createOrder(input: CreateOrderInput!): Order!
    cancelOrder(id: ID!): Order!
    assignCourier(orderId: ID!, courierId: ID!): Order!
    updateOrderStatus(orderId: ID!, status: OrderStatus!): Order!
}

type Subscription {
    # живой трекинг: «где мой заказ» — пуш клиенту при смене статуса
    orderStatusChanged(orderId: ID!): Order!
}
```

**Что проговорить:**
- `!` = non-null; nullable-поля (`courier`, `description`, `dish(id)`) — осознанный контракт «может не быть», а не небрежность.
- `menu(filter: DishFilter)` — фильтр как input-объект; дефолт значения в схеме (`onlyAvailable: Boolean = false`).
- Мутации — только через `input` (типы-модели в аргументах запрещены).
- `Subscription.orderStatusChanged` — уйти от polling: клиент подписан, сервер пушит при смене статуса (связь с HTTP vs WS).
- N+1: `ordersByStatus { items { dish } }` — резолвер на каждый заказ → DataLoader по dishId.
- Ошибки: всегда HTTP 200 + errors[]; бизнес-ошибка отмены доставленного — в errors с extensions.code.
- Защита: MaxQueryDepth/MaxQueryComplexity; интроспекция off в проде.

## Задание 3. Что значит `ID!`, `String`, `String!`

Разбор по частям — это вопрос про **систему типов GraphQL**:

| Запись | Читается | Смысл |
|---|---|---|
| `ID` | скаляр-идентификатор | уникальный ключ (строка или число); клиент **не парсит** его и не показывает как текст; сериализуется в строку |
| `String` | UTF-8 строка **nullable** | значение может отсутствовать → в JSON ключ может быть null/отсутствовать |
| `String!` | non-null строка | значение **обязательно**; null запрещён схемой — запрос с таким полем не пройдёт валидацию, если сервер вернёт null |

- `!` — **модификатор Non-Null**. Без него тип nullable по умолчанию.
- Списки читаются **«изнутри наружу»**:
  - `[String!]!` — список не null **и** элементы не null;
  - `[String]!` — сам список не null, но элементы могут быть null;
  - `[String!]` — список может быть null, элементы нет.
- Практический смысл: nullable в схеме — **документированное** «может не быть» (`courier: Courier` у заказа), `!` — контракт обязательности (`id: ID!`). Это валидация на уровне схемы ДО исполнения — потому переименование/смена типа поля ломает клиента громко (см. карточку [GraphQL] про isbn).

## Задание 4. HTTP vs WebSocket

**Базовое отличие одной фразой:** HTTP — запрос-ответ (инициатор всегда клиент); WebSocket — постоянное **полнодуплексное** соединение: после рукопожатия обе стороны шлют данные когда угодно.

| Аспект | HTTP | WebSocket |
|---|---|---|
| Модель | request/response | полнодуплексный канал |
| Соединение | на каждый запрос (keep-alive переиспользует TCP, семантика та же) | рукопожатие один раз: `GET` + `Upgrade: websocket` → `101 Switching Protocols`, дальше свой фреймовый протокол поверх TCP |
| Кто инициирует | только клиент | обе стороны (настоящий server push) |
| Оверхед | заголовки на каждый запрос (сотни байт — КБ) | фреймы 2–14 байт |
| Формат | текст: заголовки + тело (любой MIME) | текстовые/бинарные фреймы; прикладной протокол свой (часто STOMP поверх — как в лекции 08) |
| Инфраструктура | зрелая: кэши, прокси, балансировка, статусы, ретраи | кэшей нет; прокси должны пропускать Upgrade; реконнекты, heartbeat, sticky-балансировка — на тебе |
| Состояние | stateless | stateful соединение → масштабирование через брокер/shared state |

**Когда HTTP:** CRUD, обычные REST-контракты, кэширование, запрос-ответ достаточен.
**Когда WebSocket:** чат, live-трекинг заказа/курьера (наш ресторан из задания 2!), биржевые тики, совместные редакторы, уведомления.

**Костыли одностороннего push на HTTP (назвать, чтобы показать глубину):** long polling (держим запрос открытым — расход коннектов, задержки), SSE (серверный push в одну сторону, автопереподключение из коробки; когда двусторонность не нужна — проще WS).

**Финал-формула:** *«REST на HTTP — для CRUD и контрактов, WebSocket — для событий, инициируемых сервером; в зрелых системах они соседствуют: REST для операций, WS/STOMP для реального времени — как в проекте: REST+GraphQL для всего, а polling можно заменить на STOMP поверх WS»*.

Компромиссы WS честно: авторизация в handshake или первом сообщении; переподключение и дедупликация событий на клиенте; stateful = сложнее горизонтальное масштабирование (нужен брокер сообщений — RabbitMQ у нас уже есть).
